# Beispielmenü. Starten mit:  ./automenu examples/menu.txt
# Das erste Menü in der Datei ist immer das Startmenü.

menu start "Automenu"
  item "Datum und Uhrzeit"     run date
  item "Wer bin ich"           run whoami; hostname
  item "Dateien im Ordner"     run ls -l
  item "Werkzeuge"             menu tools
  item "Über dieses Menü"      info Automenu 0.2\nEin Menü wie damals.
  item "Beenden"               exit

menu tools "Werkzeuge"
  item "Speicherplatz"         run df -h
  item "Laufende Programme"    run ps aux | head -20
  item "Zurück"                back
